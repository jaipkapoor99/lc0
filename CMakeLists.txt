# ------------------------------------------------------------------------------
# Protobuf Code Generation (Python script: scripts/compile_proto.py)
# ------------------------------------------------------------------------------
find_package(Python3 REQUIRED COMPONENTS Interpreter)

function(generate_lc0_proto PROTO_FILE OUTPUT_HEADER)
  get_filename_component(PROTO_NAME ${PROTO_FILE} NAME_WE)
  set(OUT_DIR "${CMAKE_CURRENT_BINARY_DIR}")
  add_custom_command(
    OUTPUT "${OUT_DIR}/proto/${PROTO_NAME}.pb.h"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/compile_proto.py"
            "--proto_path=${CMAKE_CURRENT_SOURCE_DIR}"
            "--cpp_out=${OUT_DIR}"
            "${CMAKE_CURRENT_SOURCE_DIR}/${PROTO_FILE}"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${PROTO_FILE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/scripts/compile_proto.py"
    COMMENT "Generating proto/${PROTO_NAME}.pb.h"
    VERBATIM
  )
  set(${OUTPUT_HEADER} "${OUT_DIR}/proto/${PROTO_NAME}.pb.h" PARENT_SCOPE)
endfunction()

generate_lc0_proto("proto/net.proto" PROTO_NET_H)
generate_lc0_proto("proto/onnx.proto" PROTO_ONNX_H)
generate_lc0_proto("proto/hlo.proto" PROTO_HLO_H)

set(ALL_PROTO_HEADERS ${PROTO_NET_H} ${PROTO_ONNX_H} ${PROTO_HLO_H})
add_custom_target(generate_proto_headers DEPENDS ${ALL_PROTO_HEADERS})
